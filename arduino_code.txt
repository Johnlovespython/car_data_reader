/*
  esp32_can_bridge.ino
  ---------------------
  Turns an ESP32 dev board + external CAN transceiver (e.g. SN65HVD230)
  into a USB-serial CAN sniffer for the SavvyPy desktop app.

  HARDWARE
  --------
  ESP32 dev board  <-->  CAN transceiver (e.g. SN65HVD230)  <-->  Car OBD-II port
      GPIO5 (TX) ------> CTX / TXD
      GPIO4 (RX) <------ CRX / RXD
      3.3V       ------> VCC
      GND        ------> GND -------------------------------> OBD-II pin 4 or 5 (GND)
                          CANH ---------------------------->  OBD-II pin 6  (CAN-H)
                          CANL ---------------------------->  OBD-II pin 14 (CAN-L)

  Change CAN_TX_PIN / CAN_RX_PIN below if you wire it differently.
  Do NOT add an extra 120-ohm termination resistor - the car already
  terminates the bus at both ends.

  SAFETY
  ------
  This sketch runs the TWAI controller in LISTEN-ONLY mode: it can
  receive frames but is physically incapable of transmitting or even
  ACK-ing on the bus. That means it cannot interfere with the car in
  any way, which is what you want for a sniffer. If you later build a
  feature that sends CAN frames (e.g. querying OBD PIDs), that needs a
  deliberate switch to TWAI_MODE_NORMAL and much more care.

  SERIAL PROTOCOL (one line per CAN frame, 115200 baud)
  -------------------------------------------------------
  <millis>,0x<id_hex>,<dlc>,<data bytes as hex, space separated>
  Example:
  156644,0x003,5,03 0F 60 1F 00
  156645,0x41B,8,FD 1A 04 FF FF FF FF FF

  Standard (11-bit) IDs print as 3 hex digits; extended (29-bit) IDs
  print as 8 hex digits, so the PC side can tell them apart from the
  ID's length/magnitude alone.

  Also emits (lines starting with a letter, not a digit):
  READY,...   once the bridge is up
  ERR,...     if something goes wrong
*/

#include "driver/twai.h"

#define CAN_TX_PIN GPIO_NUM_5
#define CAN_RX_PIN GPIO_NUM_4

// Most vehicle HS-CAN diagnostic buses run at 500 kbit/s.
// If your car is one of the few running 250 kbit/s (some older
// trucks/older MS-CAN), swap this line for TWAI_TIMING_CONFIG_250KBITS().
#define CAN_TIMING_CONFIG TWAI_TIMING_CONFIG_500KBITS()

void sendFrameOverSerial(const twai_message_t &msg);

void setup() {
  Serial.begin(115200);
  delay(300);

  twai_general_config_t g_config =
      TWAI_GENERAL_CONFIG_DEFAULT(CAN_TX_PIN, CAN_RX_PIN, TWAI_MODE_LISTEN_ONLY);
  twai_timing_config_t t_config = CAN_TIMING_CONFIG;
  twai_filter_config_t f_config = TWAI_FILTER_CONFIG_ACCEPT_ALL();

  if (twai_driver_install(&g_config, &t_config, &f_config) != ESP_OK) {
    Serial.println("ERR,TWAI driver install failed");
    while (1) delay(1000);
  }

  if (twai_start() != ESP_OK) {
    Serial.println("ERR,TWAI start failed");
    while (1) delay(1000);
  }

  Serial.println("READY,ESP32 CAN bridge online (listen-only, 500kbps)");
}

void loop() {
  twai_message_t message;

  // Block up to 100ms waiting for a frame; keeps the loop responsive
  // without busy-spinning the CPU.
  if (twai_receive(&message, pdMS_TO_TICKS(100)) == ESP_OK) {
    sendFrameOverSerial(message);
  }

  // If the bus goes error-passive/bus-off (e.g. wiring issue, wrong
  // baud rate), try to recover automatically instead of hanging.
  twai_status_info_t status;
  twai_get_status_info(&status);
  if (status.state == TWAI_STATE_BUS_OFF) {
    Serial.println("ERR,Bus-off, attempting recovery");
    twai_initiate_recovery();
    delay(500);
  }
}

void sendFrameOverSerial(const twai_message_t &msg) {
  // Format: <millis>,0x<ID hex>,<dlc>,<data bytes hex, space separated>
  // Standard (11-bit) IDs are printed as 3 hex digits, extended (29-bit)
  // IDs as 8 hex digits, e.g. 0x003 vs 0x1FFFFFFF - this lets the PC
  // side tell them apart just by looking at the ID's magnitude.
  char idBuf[9];
  if (msg.extd) {
    sprintf(idBuf, "%08lX", (unsigned long)msg.identifier);
  } else {
    sprintf(idBuf, "%03lX", (unsigned long)msg.identifier);
  }

  String line = String(millis());
  line += ",0x";
  line += idBuf;
  line += ",";
  line += String(msg.data_length_code);
  line += ",";

  char byteBuf[3];
  for (int i = 0; i < msg.data_length_code; i++) {
    sprintf(byteBuf, "%02X", msg.data[i]);
    line += byteBuf;
    if (i < msg.data_length_code - 1) line += " ";
  }

  Serial.println(line);
}
